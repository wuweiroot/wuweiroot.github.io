    PAR0,   32,
    PAR1,   32,
    PAR2,   32,
    PAR3,   32,

Method( SMI, 5 , Serialized) {  //定义了SMI方法接受5个参数
    Acquire( MSMI, 0xFFFF )        // Start NVS access
    Store( ARG0, CMD )             // Set command ID
    Store( 1, ERR )                // Set pending error         //@RETRYSMI
    Store( ARG1, PAR0 )            // Set Parameter
    Store( ARG2, PAR1 )
    Store( ARG3, PAR2 )
    Store( ARG4, PAR3 )
    Store( CONFIG_ASLSMI_COMMAND, APMC ) // Genarate SMI
    While( LEqual( ERR, 1 ) ) {    // Retry if pending error
//@RETRYSMI            Sleep( 100 )
        Sleep( 1 )                                              //@RETRYSMI
        Store( CONFIG_ASLSMI_COMMAND, APMC )
    }
    Store( PAR0, Local0 )          //
    Release( MSMI )                // End of NVS access
    Return( Local0 )               // Return data
}
